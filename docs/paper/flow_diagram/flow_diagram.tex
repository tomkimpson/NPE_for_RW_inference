\documentclass[tikz,border=10pt]{standalone}
\usetikzlibrary{
	shapes.geometric, % For ellipse, etc.
	arrows.meta,      % For arrow tips
	positioning,      % For node placement (of=, below=, etc.)
	fit,              % To fit nodes around other nodes
	backgrounds,      % To draw on a background layer
	shadows.blur,     % For soft shadows
	calc              % For coordinate calculations
}

\begin{document}
	
	% --- Define a modern color palette ---
	\colorlet{simcolor}{cyan!60!black!15}
	\colorlet{datacolor}{green!60!black!15}
	\colorlet{netcolor}{orange!80!black!20}
	\colorlet{postcolor}{purple!60!black!20}
	\colorlet{phasecolor}{gray!10}
	
	% --- Define custom styles for consistency and aesthetics ---
	\tikzset{
		base/.style={
			draw,
			text width=6.5em,
			text centered,
			minimum height=3.5em,
			blur shadow={shadow opacity=25, shadow yshift=-1pt} % Add a subtle shadow
		},
		process/.style={
			base,
			rectangle,
			rounded corners=8pt, % Softer corners
			fill=simcolor
		},
		data/.style={
			base,
			ellipse,
			fill=datacolor,
			text width=5.5em,
			minimum height=3em
		},
		network/.style={
			base,
			rectangle,
			rounded corners=8pt,
			fill=netcolor,
			minimum height=6em,
			font=\bfseries
		},
		posterior/.style={
			base,
			ellipse,
			fill=postcolor,
			font=\bfseries
		},
		arrow/.style={
			-{Stealth[length=10pt, width=7pt]}, % Nicer arrow tip
			thick,
			line cap=round
		},
		phaselabel/.style={
			font=\small\bfseries,
			text=black!60
		}
	}
	
	\begin{tikzpicture}[node distance=1.5cm and 2.5cm]
		
		% --- Column 1: Simulation Process ---
		\node[process] (prior) {Prior \\ $p(\theta)$};
		\node[data, below=1.5cm of prior] (simulator) {Simulator \\ $p(x|\theta)$};
		
		% Arrow between prior and simulator
		\draw[arrow] (prior) -- (simulator) node[midway, right, font=\small] {sample $\theta$};
		
		% Group prior and simulator into a logical block
		\node[draw, dashed, rounded corners=10pt, fit={(prior) (simulator)}, inner sep=10pt] (sim_block) {};
		
		
		% --- Column 2: Simulated Data ---
		\node[data, right=of sim_block] (sim_data) {Simulated Data \\ $\{(\theta_i, x_i)\}_{i=1}^N$};
		
		
		% --- Column 3: Neural Network ---
		\node[network, right=of sim_data] (nn) {Neural Density Estimator \\ $q_\phi(\theta|x)$};
		
		
		% --- Column 4: Inference ---
		\node[data, above=1cm of nn] (obs_data) {Observed Data \\ $x_o$};
		\node[posterior, right=of nn] (posterior) {Posterior \\ $p(\theta|x_o)$};
		
		
		% --- Arrows connecting the workflow ---
		\draw[arrow] (sim_block) -- (sim_data) node[midway, above, font=\small] {Simulate $N$ times};
		\draw[arrow] (sim_data) -- (nn) node[midway, above, font=\small] {Train Network};
		\draw[arrow] (obs_data) -- (nn);
		\draw[arrow] (nn) -- (posterior) node[midway, above, font=\small] {Evaluate};
		
		
		% --- Background layers for phases ---
		\begin{scope}[on background layer]
			% Training Phase Box
			\node[
			fill=phasecolor,
			fit={(sim_block) (sim_data) (nn)},
			inner sep=20pt,
			rounded corners=15pt,
			label={[phaselabel, anchor=north west]north west:TRAINING PHASE}
			] (training_bg) {};
			
			% Inference Phase Box
			\node[
			fill=phasecolor!60!white,
			fit={(obs_data) (posterior) ($(nn.north east)+(0,0.5)$) ($(nn.south east)+(-1.2,0)$)},
			inner sep=20pt,
			rounded corners=15pt,
			label={[phaselabel, anchor=north east]north east:INFERENCE PHASE}
			] (inference_bg) {};
		\end{scope}
		
	\end{tikzpicture}
	
\end{document}
